# Sync Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement automatic file sync — Local→Remote via a `chokidar` file watcher in the `sync` app, and Remote→Local via an SSE stream with polling fallback.

**Architecture:** The `sync` app watches `sync-folder/` at the repo root. On local change it computes SHA-256, compares with cached checksum, and triggers the same chunked upload flow as the web app. For Remote→Local, it connects to `GET /sync/events` (SSE). If disconnected, it polls `GET /sync/changes?since=<timestamp>` every 30 seconds. The API writes to `sync_log` on every file mutation and broadcasts SSE to all connected clients for that user.

**Tech Stack:** `chokidar`, `eventsource`, `@aws-sdk/client-s3`, Node.js, TypeScript, Express SSE

**Prerequisites:** Plans 1, 2, and 3 complete.

```typescript
import { Router, Response } from 'express';
import { pool } from '../db';
import { requireAuth, AuthRequest } from '../middleware/auth';

export const syncRouter = Router();

// Map of userId → set of SSE response objects
const clients = new Map<string, Set<Response>>();

export function broadcastSyncEvent(userId: string, event: { fileId: string; eventType: string }) {
  const userClients = clients.get(userId);
  if (!userClients) return;
  const data = JSON.stringify(event);
  userClients.forEach((res) => res.write(`data: ${data}\n\n`));
}

// SSE stream
syncRouter.get('/events', requireAuth, (req: AuthRequest, res) => {
  res.setHeader('Content-Type', 'text/event-stream');
  res.setHeader('Cache-Control', 'no-cache');
  res.setHeader('Connection', 'keep-alive');
  res.flushHeaders();

  const userId = req.userId!;
  if (!clients.has(userId)) clients.set(userId, new Set());
  clients.get(userId)!.add(res);

  // Heartbeat every 30s to keep connection alive
  const heartbeat = setInterval(() => res.write(': heartbeat\n\n'), 30000);

  req.on('close', () => {
    clearInterval(heartbeat);
    clients.get(userId)?.delete(res);
    if (clients.get(userId)?.size === 0) clients.delete(userId);
  });
});

// Polling fallback
syncRouter.get('/changes', requireAuth, async (req: AuthRequest, res) => {
  const since = req.query.since as string || new Date(0).toISOString();

  const { rows } = await pool.query(
    'SELECT file_id, event_type, created_at FROM sync_log WHERE user_id=$1 AND created_at > $2 ORDER BY created_at ASC',
    [req.userId, since]
  );
  return res.json(rows);
});
```

```typescript
import fs from 'fs';
import path from 'path';

const STATE_FILE = path.join(__dirname, '..', '.sync-state.json');

interface SyncState {
  checksums: Record<string, string>;  // filePath → SHA-256
  lastSyncAt: string;                 // ISO timestamp
}

function load(): SyncState {
  try {
    return JSON.parse(fs.readFileSync(STATE_FILE, 'utf8'));
  } catch {
    return { checksums: {}, lastSyncAt: new Date(0).toISOString() };
  }
}

function save(state: SyncState) {
  fs.writeFileSync(STATE_FILE, JSON.stringify(state, null, 2));
}

export function getChecksum(filePath: string): string | undefined {
  return load().checksums[filePath];
}

export function setChecksum(filePath: string, checksum: string) {
  const state = load();
  state.checksums[filePath] = checksum;
  save(state);
}

export function removeChecksum(filePath: string) {
  const state = load();
  delete state.checksums[filePath];
  save(state);
}

export function getLastSyncAt(): string {
  return load().lastSyncAt;
}

export function setLastSyncAt(ts: string) {
  const state = load();
  state.lastSyncAt = ts;
  save(state);
}
```

```typescript
import fs from 'fs';
import path from 'path';
import crypto from 'crypto';
import axios from 'axios';
import { setChecksum } from './state';

const API_URL = process.env.API_URL || 'http://localhost:3000';
const TOKEN = process.env.SYNC_TOKEN || '';
const CHUNK_SIZE = 10 * 1024 * 1024; // 10MB

function computeChecksum(filePath: string): string {
  const hash = crypto.createHash('sha256');
  hash.update(fs.readFileSync(filePath));
  return hash.digest('hex');
}

function splitFile(filePath: string): Buffer[] {
  const buf = fs.readFileSync(filePath);
  const chunks: Buffer[] = [];
  for (let i = 0; i < buf.length; i += CHUNK_SIZE) {
    chunks.push(buf.slice(i, i + CHUNK_SIZE));
  }
  return chunks.length > 0 ? chunks : [Buffer.alloc(0)];
}

const api = axios.create({
  baseURL: API_URL,
  headers: { Authorization: `Bearer ${TOKEN}` },
});

export async function uploadFile(filePath: string, fileName: string): Promise<void> {
  const stat = fs.statSync(filePath);
  const checksum = computeChecksum(filePath);
  const chunks = splitFile(filePath);

  const { data: initData } = await api.post('/upload/init', {
    fileName,
    mimeType: 'application/octet-stream',
    sizeBytes: stat.size,
    totalChunks: chunks.length,
    checksum,
  });

  const { uploadId, chunkUrls } = initData as { uploadId: string; chunkUrls: string[] };
  const parts: { partNumber: number; eTag: string }[] = [];

  for (let i = 0; i < chunks.length; i++) {
    const res = await axios.put(chunkUrls[i], chunks[i], {
      headers: { 'Content-Type': 'application/octet-stream' },
    });
    parts.push({ partNumber: i + 1, eTag: res.headers.etag as string });
    console.log(`[sync] uploaded chunk ${i + 1}/${chunks.length} of ${fileName}`);
  }

  await api.post('/upload/complete', { uploadId, parts });
  setChecksum(filePath, checksum);
  console.log(`[sync] uploaded ${fileName}`);
}
```

```typescript
import fs from 'fs';
import path from 'path';
import axios from 'axios';
import { setChecksum } from './state';

const API_URL = process.env.API_URL || 'http://localhost:3000';
const TOKEN = process.env.SYNC_TOKEN || '';
const SYNC_FOLDER = path.join(process.cwd(), 'sync-folder');

const api = axios.create({
  baseURL: API_URL,
  headers: { Authorization: `Bearer ${TOKEN}` },
});

export async function downloadFile(fileId: string): Promise<void> {
  // Get file metadata
  const { data: fileMeta } = await api.get(`/files/${fileId}`);
  const destPath = path.join(SYNC_FOLDER, fileMeta.name);

  // Get presigned download URL
  const { data: urlData } = await api.get(`/files/${fileId}/download-url`);

  // Stream download to disk
  const response = await axios.get(urlData.url, { responseType: 'stream' });
  const writer = fs.createWriteStream(destPath);

  await new Promise<void>((resolve, reject) => {
    response.data.pipe(writer);
    writer.on('finish', resolve);
    writer.on('error', reject);
  });

  if (fileMeta.checksum) {
    setChecksum(destPath, fileMeta.checksum);
  }

  console.log(`[sync] downloaded ${fileMeta.name}`);
}

export async function deleteLocalFile(fileId: string): Promise<void> {
  // We don't know filename from fileId at this point — scan sync-folder
  // for a file whose checksum matches (handled by watcher state)
  // For simplicity, fetch metadata first (file may already be deleted from DB)
  try {
    const { data: fileMeta } = await api.get(`/files/${fileId}`);
    const destPath = path.join(SYNC_FOLDER, fileMeta.name);
    if (fs.existsSync(destPath)) {
      fs.unlinkSync(destPath);
      console.log(`[sync] deleted local ${fileMeta.name}`);
    }
  } catch {
    // File already gone from DB — ignore
  }
}
```

```typescript
import chokidar from 'chokidar';
import crypto from 'crypto';
import fs from 'fs';
import path from 'path';
import { getChecksum, setChecksum, removeChecksum } from './state';
import { uploadFile } from './uploader';

const SYNC_FOLDER = path.join(process.cwd(), 'sync-folder');
const DEBOUNCE_MS = 500;

const debounceTimers = new Map<string, ReturnType<typeof setTimeout>>();

function scheduleUpload(filePath: string) {
  const existing = debounceTimers.get(filePath);
  if (existing) clearTimeout(existing);

  const timer = setTimeout(async () => {
    debounceTimers.delete(filePath);
    try {
      const buf = fs.readFileSync(filePath);
      const checksum = crypto.createHash('sha256').update(buf).digest('hex');
      const cached = getChecksum(filePath);
      if (checksum === cached) return; // no real change

      const fileName = path.basename(filePath);
      await uploadFile(filePath, fileName);
      setChecksum(filePath, checksum);
    } catch (err) {
      console.error(`[watcher] upload failed for ${filePath}:`, err);
    }
  }, DEBOUNCE_MS);

  debounceTimers.set(filePath, timer);
}

export function startWatcher() {
  const watcher = chokidar.watch(SYNC_FOLDER, {
    ignoreInitial: false,
    ignored: /(^|[/\\])\../,  // ignore dotfiles
  });

  watcher
    .on('add', (filePath) => {
      console.log(`[watcher] add: ${path.basename(filePath)}`);
      scheduleUpload(filePath);
    })
    .on('change', (filePath) => {
      console.log(`[watcher] change: ${path.basename(filePath)}`);
      scheduleUpload(filePath);
    })
    .on('unlink', async (filePath) => {
      console.log(`[watcher] delete: ${path.basename(filePath)}`);
      removeChecksum(filePath);
      // Note: deletion of remote file on local delete is a product decision —
      // for safety we only sync local→remote for adds/changes, not deletes.
    });

  console.log(`[watcher] watching ${SYNC_FOLDER}`);
  return watcher;
}
```

- [ ] **Step 3: Create apps/sync/src/index.ts**

```typescript
import EventSource from 'eventsource';
import { startWatcher } from './watcher';
import { downloadFile, deleteLocalFile } from './downloader';
import { getLastSyncAt, setLastSyncAt } from './state';
import axios from 'axios';

const API_URL = process.env.API_URL || 'http://localhost:3000';
const TOKEN = process.env.SYNC_TOKEN || '';
const POLL_INTERVAL_MS = 30000;

const api = axios.create({
  baseURL: API_URL,
  headers: { Authorization: `Bearer ${TOKEN}` },
});

async function processSyncEvent(event: { file_id: string; event_type: string }) {
  if (event.event_type === 'created' || event.event_type === 'updated') {
    await downloadFile(event.file_id);
  } else if (event.event_type === 'deleted') {
    await deleteLocalFile(event.file_id);
  }
  setLastSyncAt(new Date().toISOString());
}

function connectSSE() {
  const url = `${API_URL}/sync/events`;
  const es = new EventSource(url, {
    headers: { Authorization: `Bearer ${TOKEN}` },
  });

  es.onmessage = async (e) => {
    try {
      const event = JSON.parse(e.data);
      await processSyncEvent(event);
    } catch (err) {
      console.error('[sse] failed to process event:', err);
    }
  };

  es.onerror = () => {
    console.warn('[sse] connection lost — falling back to polling');
    es.close();
    // Reconnect after 5s
    setTimeout(connectSSE, 5000);
  };

  console.log('[sse] connected');
  return es;
}

async function poll() {
  const since = getLastSyncAt();
  try {
    const { data } = await api.get(`/sync/changes?since=${since}`);
    for (const event of data) {
      await processSyncEvent(event);
    }
  } catch (err) {
    console.error('[poll] failed:', err);
  }
}

async function main() {
  if (!TOKEN) {
    console.error('SYNC_TOKEN env var required. Get it by logging in via the web app and copying the JWT.');
    process.exit(1);
  }

  console.log('[sync] starting PJDrive sync client');

  // Catch up on missed changes before connecting SSE
  await poll();

  // Start file watcher (Local → Remote)
  startWatcher();

  // Connect SSE (Remote → Local)
  connectSSE();

  // Polling fallback runs every 30s as belt-and-suspenders
  setInterval(poll, POLL_INTERVAL_MS);
}

main().catch((err) => { console.error(err); process.exit(1); });
```
